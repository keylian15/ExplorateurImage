    main_window

    views.main_window

    Fenêtre principale de l'application — navigateur à onglets.

    Chaque onglet Galerie ou Carte 2D est entièrement autonome : il possède ses propres instances de GalleryViewModel, DetailViewModel, AutocompleteViewModel et MapViewModel.

    Seuls le thème (StyleTab) et la configuration (config) sont partagés entre tous les onglets.

    Responsabilités
    1. Gérer un système d'onglets de type navigateur (déplaçables, fermables)
    2. Instancier les ViewModels de manière isolée par onglet
    3. Afficher un onglet "Nouvel onglet" avec 3 boutons de création
    4. Maintenir un seul onglet de thème global partagé
    5. Gérer l'ouverture de dossiers et la coordination interne à chaque onglet

    class GalleryTabContainer(QWidget):
    """Onglet Galerie autonome avec ses propres ViewModels et dock de détail.

    Chaque instance est totalement indépendante : ses ViewModels, son cache,
    son index et son dock de détail sont propres à cet onglet.
    """

    def __init__(self, client: OllamaWrapper, config: dict, parent=None):
        """
        Args:
            client (OllamaWrapper): Client Ollama partagé.
            config (dict): Configuration globale partagée.
            parent (QWidget, optional): Parent. Defaults to None.
        """
        super().__init__(parent)

        # ── ViewModels propres à cet onglet ───────────────────────────────────
        self.gallery_vm = GalleryViewModel(client, config)
        self.detail_vm = DetailViewModel(client, config, self.gallery_vm)
        self.autocomplete_vm = AutocompleteViewModel(client, self.gallery_vm)

        # ── Widgets ───────────────────────────────────────────────────────────
        self._gallery_widget = GalleryWidget(self.gallery_vm, self.autocomplete_vm, self)
        self._detail_widget = DetailWidget(self.detail_vm, self)

        # Câble le bouton "Ouvrir" localement
        self._gallery_widget.btn_open.clicked.connect(self.open_folder_dialog)

        # ── Dock de détail propre à cet onglet ────────────────────────────────
        # Le dock est intégré directement dans le layout (pas dans un QMainWindow)
        # On utilise une disposition horizontale : galerie à gauche, détail à droite.
        self._detail_panel = QWidget()
        self._detail_panel.setFixedWidth(300)
        self._detail_panel.setVisible(False)

        detail_layout = QVBoxLayout(self._detail_panel)
        detail_layout.setContentsMargins(0, 0, 0, 0)

        # Barre de titre du panneau de détail
        title_bar = QHBoxLayout()
        lbl_title = QLabel("Détails")
        lbl_title.setStyleSheet(detail_label_style())
        btn_close = QPushButton("✕")
        btn_close.setFixedSize(24, 24)
        btn_close.setStyleSheet(detail_btn_style())
        btn_close.clicked.connect(lambda: self._detail_panel.setVisible(False))
        title_bar.addWidget(lbl_title)
        title_bar.addStretch()
        title_bar.addWidget(btn_close)
        detail_layout.addLayout(title_bar)
        detail_layout.addWidget(self._detail_widget)

        # ── Layout principal ──────────────────────────────────────────────────
        h_layout = QHBoxLayout(self)
        h_layout.setContentsMargins(0, 0, 0, 0)
        h_layout.setSpacing(0)
        h_layout.addWidget(self._gallery_widget, stretch=1)
        h_layout.addWidget(self._detail_panel)

        # ── Connexions ────────────────────────────────────────────────────────
        self.gallery_vm.image_selected.connect(self.on_image_selected)

    def open_folder_dialog(self):
        """Ouvre un sélecteur de dossier et charge les images."""
        folder = QFileDialog.getExistingDirectory(self, "Choisir un dossier")
        if folder:
            self.gallery_vm.open_folder(folder)

    def on_image_selected(self, img_name: str):
        """Affiche le panneau de détail lors de la sélection d'une image.

        Args:
            img_name (str): Nom de l'image sélectionnée.
        """
        if not self._detail_panel.isVisible():
            self._detail_panel.setVisible(True)
        self.detail_vm.on_image_selected(img_name)

    def tab_label(self) -> str:
        """Retourne le libellé à afficher dans l'onglet.

        Returns:
            str: Nom du dossier courant ou libellé par défaut.
        """
        if self.gallery_vm.current_folder:
            import os

            return f"🖼  {os.path.basename(self.gallery_vm.current_folder)}"
        return "🖼  Galerie"

    class MainWindow(QMainWindow):
    """Fenêtre principale agissant comme un navigateur à onglets.

    Chaque onglet Galerie ou Carte 2D est autonome (ViewModels isolés).
    Le thème et la configuration sont partagés entre tous les onglets.
    Un onglet "+" permet d'en créer de nouveaux via NewTabPage.
    """

    # Index réservé pour l'onglet "+" (toujours en dernière position)
    _PLUS_TAB_LABEL = "+"

    def __init__(
        self,
        client: OllamaWrapper,
        config: dict,
    ):
        """
        Args:
            client (OllamaWrapper): Client Ollama partagé entre tous les onglets.
            config (dict): Configuration globale partagée.
        """
        super().__init__()
        self._client = client
        self._config = config

        self.setWindowTitle("Explorateur d'images")
        screen = QApplication.primaryScreen().availableGeometry()
        self.setGeometry(screen)

        self.build_ui()

        # Ouvre un premier onglet Galerie au démarrage
        self.create_gallery_tab()

        # Charge le dossier par défaut dans le premier onglet galerie
        default_folder = config.get("default_folder")
        if default_folder:
            import os

            if os.path.exists(default_folder):
                first = self.tabs.widget(0)
                if isinstance(first, GalleryTabContainer):
                    first.gallery_vm.open_folder(default_folder)
                    self.tabs.setTabText(0, first.tab_label())

    # ── Construction ──────────────────────────────────────────────────────────

    def build_ui(self):
        """Construit l'interface de la fenêtre principale."""

        # Onglet thème partagé (unique)
        self._style_tab = StyleTab(self)

        # Système d'onglets principal
        self.tabs = QTabWidget()
        self.tabs.setMovable(True)
        self.tabs.setTabsClosable(True)
        self.tabs.setDocumentMode(True)

        # Ajoute l'onglet "+" fixe en fin de liste
        self._plus_index = self.tabs.addTab(QWidget(), self._PLUS_TAB_LABEL)
        # Le "+" n'est pas fermable
        self.tabs.tabBar().setTabButton(self._plus_index, self.tabs.tabBar().ButtonPosition.RightSide, None)

        self.tabs.tabCloseRequested.connect(self.close_tab)
        self.tabs.currentChanged.connect(self.on_tab_changed)

        self.setCentralWidget(self.tabs)

    # ── Création d'onglets ────────────────────────────────────────────────────

    def insert_tab(self, widget: QWidget, label: str) -> int:
        """Insère un onglet juste avant le "+" et le sélectionne.

        Args:
            widget (QWidget): Contenu de l'onglet.
            label (str): Libellé de l'onglet.

        Returns:
            int: Index de l'onglet créé.
        """
        plus_idx = self.plus_tab_index()
        idx = self.tabs.insertTab(plus_idx, widget, label)
        self.tabs.setCurrentIndex(idx)
        return idx

    def create_gallery_tab(self) -> GalleryTabContainer:
        """Crée et insère un nouvel onglet Galerie autonome.

        Returns:
            GalleryTabContainer: Le conteneur créé.
        """
        container = GalleryTabContainer(self._client, self._config, self)

        # Met à jour le libellé de l'onglet quand le dossier change
        container.gallery_vm.folder_changed.connect(lambda _: self.update_tab_label(container))

        _idx = self.insert_tab(container, container.tab_label())
        # L'onglet "+" reste non-fermable (recalcul après insertion)
        self.refresh_plus_close_button()
        return container

    def create_map_tab(self) -> MapTabContainer:
        """Crée et insère un nouvel onglet Carte 2D autonome.

        Returns:
            MapTabContainer: Le conteneur créé.
        """
        container = MapTabContainer(self._client, self._config, self, self)
        self.insert_tab(container, container.tab_label())
        self.refresh_plus_close_button()
        return container

    def create_theme_tab(self):
        """Affiche l'onglet Thème. S'il est déjà ouvert, le sélectionne."""
        # Cherche si le StyleTab est déjà ouvert
        for i in range(self.tabs.count()):
            if isinstance(self.tabs.widget(i), StyleTab):
                self.tabs.setCurrentIndex(i)
                return

        idx = self.insert_tab(self._style_tab, "🎨  Thème")
        self.refresh_plus_close_button()
        self.tabs.setCurrentIndex(idx)

    # ── Gestion des onglets ───────────────────────────────────────────────────

    def on_tab_changed(self, index: int):
        """Intercepte le clic sur "+" pour ouvrir un NewTabPage.

        Args:
            index (int): Index de l'onglet sélectionné.
        """
        if self.tabs.tabText(index) == self._PLUS_TAB_LABEL:
            # Remplace le contenu du "+" par une page de démarrage
            page = NewTabPage(
                on_gallery=self.create_gallery_tab,
                on_map=self.create_map_tab,
                on_theme=self.create_theme_tab,
            )
            # Insère la page avant "+" et sélectionne-la
            plus_idx = self.plus_tab_index()
            new_idx = self.tabs.insertTab(plus_idx, page, "Nouvel onglet")
            self.tabs.setCurrentIndex(new_idx)
            self.refresh_plus_close_button()

    def close_tab(self, index: int):
        """Ferme l'onglet à l'index donné.

        Interdit la fermeture du "+" et du dernier onglet de contenu.

        Args:
            index (int): Index de l'onglet à fermer.
        """
        if self.tabs.tabText(index) == self._PLUS_TAB_LABEL:
            return

        # Compte les onglets de contenu (hors "+")
        content_count = sum(1 for i in range(self.tabs.count()) if self.tabs.tabText(i) != self._PLUS_TAB_LABEL)
        if content_count <= 1:
            return

        widget = self.tabs.widget(index)
        self.tabs.removeTab(index)
        if widget and not isinstance(widget, StyleTab):
            widget.deleteLater()

    def update_tab_label(self, container: GalleryTabContainer | MapTabContainer):
        """Met à jour le libellé de l'onglet correspondant au conteneur.

        Args:
            container: Conteneur dont le libellé doit être mis à jour.
        """
        for i in range(self.tabs.count()):
            if self.tabs.widget(i) is container:
                self.tabs.setTabText(i, container.tab_label())
                break

    def plus_tab_index(self) -> int:
        """Retourne l'index de l'onglet "+".

        Returns:
            int: Index de l'onglet "+".
        """
        for i in range(self.tabs.count()):
            if self.tabs.tabText(i) == self._PLUS_TAB_LABEL:
                return i
        return self.tabs.count()

    def refresh_plus_close_button(self):
        """S'assure que l'onglet "+" n'a pas de bouton de fermeture."""
        plus_idx = self.plus_tab_index()
        if plus_idx < self.tabs.count():
            self.tabs.tabBar().setTabButton(
                plus_idx,
                self.tabs.tabBar().ButtonPosition.RightSide,
                None,
            )

    def on_tab_changed(self, index: int):
    """Intercepte le clic sur "+" pour ouvrir un NewTabPage.

    Args:
        index (int): Index de l'onglet sélectionné.
    """
    if self.tabs.tabText(index) == self._PLUS_TAB_LABEL:
        # Remplace le contenu du "+" par une page de démarrage
        page = NewTabPage(
            on_gallery=self.create_gallery_tab,
            on_map=self.create_map_tab,
            on_theme=self.create_theme_tab,
        )
        # Insère la page avant "+" et sélectionne-la
        plus_idx = self.plus_tab_index()
        new_idx = self.tabs.insertTab(plus_idx, page, "Nouvel onglet")
        self.tabs.setCurrentIndex(new_idx)
        self.refresh_plus_close_button()

        return btn
